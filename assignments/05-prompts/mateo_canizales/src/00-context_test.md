Estrutura clara

Exemplo do formato utilizado:

// Classe que representa um objeto do mundo real
public class Pessoa {

    // Atributos encapsulados (private)
    private String nome;
    private int idade;

    // Construtor da classe
    public Pessoa(String nome, int idade) {
        this.nome = nome;
        this.idade = idade;
    }

    // Getter para acessar o nome
    public String getNome() {
        return nome;
    }

    // Setter para modificar o nome
    public void setNome(String nome) {
        this.nome = nome;
    }

    // Método de comportamento do objeto
    public void apresentar() {
        System.out.println("Olá, meu nome é " + nome + " e tenho " + idade + " anos.");
    }
}
